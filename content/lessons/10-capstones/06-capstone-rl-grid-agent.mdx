## Project Brief: Reliable Policy Evaluation Under Stochasticity

In Reinforcement Learning, high training reward can hide brittle luck if evaluation leaves exploration noise turned on or tests only one fixed seed! In Capstone 5, you build a complete 5x5 stochastic GridWorld simulator, decay exploration smoothly across training episodes, and verify the greedy learned policy ($\varepsilon = 0$) achieves consistent optimal goal completion.

  code={`import numpy as np

class GridEnv:
    def __init__(self):
        self.walls = {(1, 1), (2, 1), (3, 3)}
        self.goal = (0, 4)
        self.pit = (1, 3)
        self.deltas = [(-1, 0), (0, 1), (1, 0), (0, -1)]

    def reset(self):
        self.pos = (4, 0)
        return self.pos

    def step(self, action):
        dr, dc = self.deltas[action]
        nr = max(0, min(4, self.pos[0] + dr))
        nc = max(0, min(4, self.pos[1] + dc))
        if (nr, nc) not in self.walls:
            self.pos = (nr, nc)
        if self.pos == self.goal:
            return self.pos, 10.0, True
        if self.pos == self.pit:
            return self.pos, -10.0, True
        return self.pos, -0.2, False

env = GridEnv()
Q = np.zeros((5, 5, 4))
rng = np.random.default_rng(42)

for ep in range(300):
    s = env.reset()
    eps = max(0.05, 0.5 * (0.99 ** ep))
    for _ in range(50):
        a = int(rng.integers(4)) if rng.random() < eps else int(np.argmax(Q[s[0], s[1]]))
        ns, r, done = env.step(a)
        target = r + (0.0 if done else 0.92 * np.max(Q[ns[0], ns[1]]))
        Q[s[0], s[1], a] += 0.25 * (target - Q[s[0], s[1], a])
        s = ns
        if done:
            break

# Greedy evaluation (epsilon=0)
s = env.reset()
steps, total_r = 0, 0.0
for _ in range(30):
    a = int(np.argmax(Q[s[0], s[1]]))
    s, r, done = env.step(a)
    total_r += r; steps += 1
    if done:
        break

print(f"Greedy Eval -> Reached={s == env.goal} in {steps} steps, Return={total_r:.1f}")`}
