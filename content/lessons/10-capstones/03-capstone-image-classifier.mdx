Accuracy alone doesn't prove a vision model learned the right visual cues—a famous husky-vs-wolf classifier actually cheated by detecting snow in the background! In this capstone, you train a compact CNN and compute **Grad-CAM** spatial attribution to verify which pixels drive the prediction.

# Generate clean 8x8 synthetic horizontal vs vertical bar dataset
def make_bars(n=40, seed=0):
    rng = np.random.default_rng(seed)
    X, y = [], []
    for i in range(n):
        img = rng.normal(0, 0.1, (8, 8))
        if i % 2 == 0:
            img[3:5, 1:7] += 1.0  # horizontal bar -> class 0
            y.append(0)
        else:
            img[1:7, 3:5] += 1.0  # vertical bar -> class 1
            y.append(1)
        X.append(img)
    return np.array(X), np.array(y)

X, y = make_bars()
# Simple horizontal vs vertical edge kernel response discriminates perfectly
h_score = np.mean(np.abs(np.diff(X, axis=1)), axis=(1, 2))
v_score = np.mean(np.abs(np.diff(X, axis=2)), axis=(1, 2))
preds = (v_score > h_score).astype(int)
print("Edge-energy baseline accuracy:", float(np.mean(preds == y)))`}
  libraryCode={`import torch
import torch.nn as nn
import torch.nn.functional as F

torch.manual_seed(7)
class MiniCNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = nn.Conv2d(1, 8, kernel_size=3, padding=1)
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.fc = nn.Linear(8, 2)

    def forward(self, x):
        feats = F.relu(self.conv(x))
        pooled = self.pool(feats).flatten(1)
        return self.fc(pooled), feats

# Train on synthetic horizontal (0) vs vertical (1) 8x8 bars
X = torch.zeros(24, 1, 8, 8)
y = torch.zeros(24, dtype=torch.long)
for i in range(24):
    if i % 2 == 0:
        X[i, 0, 3:5, 1:7] = 1.0; y[i] = 0
    else:
        X[i, 0, 1:7, 3:5] = 1.0; y[i] = 1

net = MiniCNN()
opt = torch.optim.AdamW(net.parameters(), lr=0.04)
for _ in range(40):
    logits, _ = net(X)
    loss = F.cross_entropy(logits, y)
    opt.zero_grad(); loss.backward(); opt.step()

net.eval()
with torch.inference_mode():
    logits, _ = net(X)
    acc = (logits.argmax(dim=1) == y).float().mean().item()
print(f"Trained MiniCNN accuracy: {acc:.1%} | Final loss: {loss.item():.4f}")`}

  exerciseSolution={`import numpy as np

def compute_gradcam_map(feature_maps, gradients):
    weights = np.mean(gradients, axis=(1, 2))
    cam = np.sum(weights[:, None, None] * feature_maps, axis=0)
    return np.maximum(0.0, cam)`}
