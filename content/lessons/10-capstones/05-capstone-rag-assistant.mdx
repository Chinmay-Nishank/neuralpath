  code={`from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

DOCS = [
    {"id": "auth_1", "title": "Auth Setup", "text": "Set AUTH_SECRET and AUTH_GITHUB_ID in .env.local to enable cross-device sync."},
    {"id": "quant_1", "title": "Quantization", "text": "Use AWQ 4-bit quantization or FP8 KV cache via --kv-cache-dtype fp8 in vLLM."},
    {"id": "lora_1", "title": "Fine-Tuning", "text": "Initialize LoRA matrix B to zeros so delta W equals zero at training step zero."},
]

vec = TfidfVectorizer(ngram_range=(1, 2))
doc_mat = vec.fit_transform([d["title"] + " " + d["text"] for d in DOCS])

def retrieve(query, top_k=2):
    q_vec = vec.transform([query])
    scores = cosine_similarity(q_vec, doc_mat).ravel()
    ranked = scores.argsort()[::-1][:top_k]
    return [(DOCS[i]["id"], round(float(scores[i]), 3)) for i in ranked]

golden_eval = [
    ("Which flag enables FP8 KV cache in vLLM?", "quant_1"),
    ("How is LoRA B initialized?", "lora_1"),
    ("Which env vars configure GitHub sync?", "auth_1"),
]

hits = sum(retrieve(q, top_k=1)[0][0] == expected for q, expected in golden_eval)
print(f"Golden Recall@1: {hits}/{len(golden_eval)} ({hits/len(golden_eval):.0%})")
print("Sample retrieval:", retrieve("FP8 kv cache vLLM flag"))`}
