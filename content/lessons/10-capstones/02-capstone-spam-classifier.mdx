  code={`import pandas as pd, math
from collections import Counter

df = pd.read_csv("/datasets/spam_mini.csv")
train_df = df.iloc[:10]
test_df = df.iloc[10:]

# Multinomial Naive Bayes with Laplace smoothing (alpha=1.0) from scratch
vocab = set()
counts = {0: Counter(), 1: Counter()}
totals = {0: 0, 1: 0}
priors = {0: (train_df["label"] == 0).mean(), 1: (train_df["label"] == 1).mean()}

for _, row in train_df.iterrows():
    words = row["text"].lower().split()
    lbl = int(row["label"])
    counts[lbl].update(words)
    totals[lbl] += len(words)
    vocab.update(words)

V = len(vocab)
def predict_nb(text):
    words = text.lower().split()
    scores = {}
    for c in (0, 1):
        s = math.log(priors[c] + 1e-9)
        for w in words:
            s += math.log((counts[c][w] + 1.0) / (totals[c] + V))
        scores[c] = s
    return 1 if scores[1] > scores[0] else 0

preds = [predict_nb(t) for t in test_df["text"]]
print("Scratch NB predictions:", preds, "| Truth:", test_df["label"].tolist())`}
  libraryCode={`import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import StratifiedKFold, cross_val_score
from sklearn.pipeline import Pipeline

df = pd.read_csv("/datasets/spam_mini.csv")
pipe = Pipeline([
    ("tfidf", TfidfVectorizer(ngram_range=(1, 2), lowercase=True, stop_words="english")),
    ("clf", LogisticRegression(C=2.0, max_iter=500)),
])

cv = StratifiedKFold(n_splits=4, shuffle=True, random_state=42)
auc = cross_val_score(pipe, df["text"], df["label"], cv=cv, scoring="roc_auc")
pipe.fit(df["text"], df["label"])

sample_probs = pipe.predict_proba(["URGENT free prize click now", "Please review the tokenizer PR"])[:, 1]
print("4-fold CV ROC-AUC:", round(float(auc.mean()), 3))
print("Test probabilities [spam, ham]:", sample_probs.round(3))`}
